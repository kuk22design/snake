<body>

    <h1>Had</h1>
    <div id="score-board">Skóre: <span id="score">0</span></div>

    <div id="game-container">
        <canvas id="gameCanvas" width="400" height="400"></canvas>
        <div id="game-over-screen" id="gameOverScreen">
            <p>Konec hry!</p>
            <button onclick="resetGame()">Hrát znovu</button>
        </div>
    </div>

    <div class="controls-info">Ovládání: Šipkami na klávesnici</div>

    <script>
        const canvas = document.getElementById("gameCanvas");
        const ctx = canvas.getContext("2d");
        const scoreElement = document.getElementById("score");
        const gameOverScreen = document.getElementById("game-over-screen");

        // Velikost jedné kostičky (mřížky)
        const gridSize = 20;
        const tileCount = canvas.width / gridSize;

        // Rychlost hry (obnovení v milisekundách)
        const gameSpeed = 100; 

        let snake = [];
        let food = { x: 0, y: 0 };
        let dx = gridSize; // Rychlost na ose X
        let dy = 0;        // Rychlost na ose Y
        let score = 0;
        let gameInterval;
        let changingDirection = false; // Ochrana proti rychlému stisknutí kláves

        // Spuštění hry
        resetGame();

        function main() {
            if (hasGameEnded()) {
                gameOverScreen.style.display = "flex";
                clearInterval(gameInterval);
                return;
            }

            changingDirection = false;
            clearCanvas();
            drawFood();
            moveSnake();
            drawSnake();
        }

        // Vyčištění plátna
        function clearCanvas() {
            ctx.fillStyle = "#222";
            ctx.fillRect(0, 0, canvas.width, canvas.height);
        }

        // Vykreslení hada
        function drawSnake() {
            snake.forEach((part, index) => {
                // Hlava je světlejší zelená, tělo tmavší
                ctx.fillStyle = index === 0 ? "#66bb6a" : "#4CAF50";
                ctx.strokeStyle = "#111";
                
                ctx.fillRect(part.x, part.y, gridSize, gridSize);
                ctx.strokeRect(part.x, part.y, gridSize, gridSize);
            });
        }

        // Pohyb hada
        function moveSnake() {
            const head = { x: snake[0].x + dx, y: snake[0].y + dy };
            snake.unshift(head);

            // Kontrola, zda had snědl jídlo
            const hasEatenFood = snake[0].x === food.x && snake[0].y === food.y;
            if (hasEatenFood) {
                score += 10;
                scoreElement.innerText = score;
                generateFood();
            } else {
                snake.pop(); // Pokud nesnědl jídlo, smaže se ocas
            }
        }

        // Generování jídla na náhodnou pozici
        function generateFood() {
            food.x = Math.floor(Math.random() * tileCount) * gridSize;
            food.y = Math.floor(Math.random() * tileCount) * gridSize;

            // Kontrola, aby se jídlo neobjevilo na hadovi
            snake.forEach(part => {
                const hasEaten = part.x === food.x && part.y === food.y;
                if (hasEaten) generateFood();
            });
        }

        // Vykreslení jídla (červené kostičky)
        function drawFood() {
            ctx.fillStyle = "#ff4444";
            ctx.fillRect(food.x, food.y, gridSize, gridSize);
        }

        // Kontrola kolizí (konec hry)
        function hasGameEnded() {
            // Kolize se sebou samým
            for (let i = 4; i < snake.length; i++) {
                if (snake[i].x === snake[0].x && snake[i].y === snake[0].y) return true;
            }

            // Kolize se stěnami
            const hitLeftWall = snake[0].x < 0;
            const hitRightWall = snake[0].x >= canvas.width;
            const hitToptWall = snake[0].y < 0;
            const hitBottomWall = snake[0].y >= canvas.height;

            return hitLeftWall || hitRightWall || hitToptWall || hitBottomWall;
        }

        // Zpracování stisknutí kláves
        document.addEventListener("keydown", changeDirection);

        function changeDirection(event) {
            const LEFT_KEY = 37;
            const UP_KEY = 38;
            const RIGHT_KEY = 39;
            const DOWN_KEY = 40;

            if (changingDirection) return;

            const keyPressed = event.keyCode;
            const goingUp = dy === -gridSize;
            const goingDown = dy === gridSize;
            const goingRight = dx === gridSize;
            const goingLeft = dx === -gridSize;

            if (keyPressed === LEFT_KEY && !goingRight) {
                dx = -gridSize;
                dy = 0;
                changingDirection = true;
            }
            if (keyPressed === UP_KEY && !goingDown) {
                dx = 0;
                dy = -gridSize;
                changingDirection = true;
            }
            if (keyPressed === RIGHT_KEY && !goingLeft) {
                dx = gridSize;
                dy = 0;
                changingDirection = true;
            }
            if (keyPressed === DOWN_KEY && !goingUp) {
                dx = 0;
                dy = gridSize;
                changingDirection = true;
            }
        }

        // Resetování hry do původního stavu
        function resetGame() {
            snake = [
                { x: 160, y: 200 },
                { x: 140, y: 200 },
                { x: 120, y: 200 }
            ];
            score = 0;
            dx = gridSize;
            dy = 0;
            scoreElement.innerText = score;
            gameOverScreen.style.display = "none";
            generateFood();
            clearInterval(gameInterval);
            gameInterval = setInterval(main, gameSpeed);
        }
    </script>
